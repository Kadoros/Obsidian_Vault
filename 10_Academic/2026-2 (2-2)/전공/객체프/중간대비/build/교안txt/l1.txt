            why use functions?
• many reasons
 – Don’t have to repeat the same block of code many times. Make
 that code block a function and call it when needed.
 – Reuse : useful functions can be reused in a number of programs.
 – top-down technique : Make an outline and hierarchy of the steps
 needed to solve your problem and create a function for each step.
 – Easy to debug : Get one function working well then move on to
 the others.
 – Easy to modify and expand : Just add more functions to extend
 program capability
 – Readabilty : Make program self-documenting and readable.

                                 10. pointer
•   pointer type variable
     – “pointer” usually means memory address
     – a variable that stores memory address
     – size is always: 8 bytes (in 64-bit systems), 4 bytes (in 32-bit systems)
int *xPtr, x;        // xPtr is pointer
                     // x is an integer type

x = 5;               // assignment
xPtr = &x;           // xPtr points to x that means
                     // the value of xPtr is the address of x

*xPtr = 10;          // the value xPtr points to is changed to 10
                     // that means x becomes 10
cout << x;           // result is 10


•   Remember
     – *
           (i) multiplication operator (binary operator), ex) z=x*y;
           (ii) the-value-of operator (unary operator), ex) *x=5;
           (iii) pointer type declaration, ex) int *x;
     – &
           (i) bitwise-AND operator (binary operator), ex) z=x&y;
           (ii) address-of operator (unary operator), ex) x=&y;
           (iii) reference type declaration, ex) int& y=x;

 #include <iostream>
 using namespace std;

 class Rectangle {
    private:
         int width, height;
    public:
         Rectangle() { width=1; height=1; }
         Rectangle(int x, int y) : width(x), height(y) {}
         int area() { return width*height; }
 };

 int main()
 {
         Rectangle obj(3,4);
         Rectangle *foo, *bar, *baz;

        foo = &obj;
        bar = new Rectangle(5,6);
        baz = new Rectangle[2];

        cout << "obj's area: " << obj.area() << '\n';
        cout << "*foo's area: " << foo->area() << '\n';
        cout << "*bar's area: " << bar->area() << '\n';
        cout << "baz[0]'s area: " << baz[0].area() << '\n';
        cout << "baz[1]'s area: " << baz[1].area() << '\n';
        delete bar;
        delete[] baz;

        return 0;
 }
